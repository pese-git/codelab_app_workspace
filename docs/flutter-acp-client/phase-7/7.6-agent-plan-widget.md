# Задача 7.6 — Agent Plan Widget

## Цель

Виджет для отображения плана выполнения агента с прогрессом по каждому пункту.

---

## Шаг 1 — Plan Panel Widget

Создай файл `lib/presentation/widgets/plan_panel.dart`:

```dart
import 'package:fluent_ui/fluent_ui.dart';
import 'package:codelab_ui_components/codelab_ui_components.dart';

import '../../domain/entities/plan_state.dart';
import '../../domain/entities/plan.dart';

class PlanPanel extends StatefulWidget {
  final PlanState plan;

  const PlanPanel({super.key, required this.plan});

  @override
  State<PlanPanel> createState() => _PlanPanelState();
}

class _PlanPanelState extends State<PlanPanel> {
  bool _isExpanded = true;

  @override
  Widget build(BuildContext context) {
    final brightness = FluentTheme.of(context).brightness;
    final colors = brightness == Brightness.light
        ? AppColors.light
        : AppColors.dark;

    final progress = widget.plan.totalCount > 0
        ? widget.plan.completedCount / widget.plan.totalCount
        : 0.0;

    return Container(
      decoration: BoxDecoration(
        color: colors.surfaceSubtle,
        border: Border(
          bottom: BorderSide(color: colors.borderBase),
        ),
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          // Заголовок с прогресс-баром
          Padding(
            padding: const EdgeInsets.symmetric(horizontal: AppSpacing.md),
            child: CollapsibleSectionHeader(
              title: 'Plan',
              subtitle: '${widget.plan.completedCount}/${widget.plan.totalCount}',
              isExpanded: _isExpanded,
              onToggle: () => setState(() => _isExpanded = !_isExpanded),
              trailing: widget.plan.totalCount > 0
                  ? SizedBox(
                      width: 80,
                      child: AppProgressBar(
                        value: progress,
                        height: 4,
                      ),
                    )
                  : null,
            ),
          ),
          // Список пунктов
          if (_isExpanded)
            ListView.builder(
              shrinkWrap: true,
              physics: const NeverScrollableScrollPhysics(),
              padding: const EdgeInsets.only(
                left: AppSpacing.md,
                right: AppSpacing.md,
                bottom: AppSpacing.sm,
              ),
              itemCount: widget.plan.entries.length,
              itemBuilder: (context, index) {
                return _PlanEntryRow(
                  entry: widget.plan.entries[index],
                );
              },
            ),
        ],
      ),
    );
  }
}

class _PlanEntryRow extends StatelessWidget {
  final PlanEntry entry;

  const _PlanEntryRow({required this.entry});

  @override
  Widget build(BuildContext context) {
    final brightness = FluentTheme.of(context).brightness;
    final colors = brightness == Brightness.light
        ? AppColors.light
        : AppColors.dark;

    final isCompleted = entry.status == PlanEntryStatus.completed;

    return Padding(
      padding: const EdgeInsets.symmetric(vertical: AppSpacing.xs2),
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          // Приоритет
          _priorityBadge(entry.priority),
          const SizedBox(width: AppSpacing.xs),
          // Статус иконка
          AppIcon(
            _iconForStatus(entry.status),
            size: AppIconSize.sm,
          ),
          const SizedBox(width: AppSpacing.xs),
          // Текст
          Expanded(
            child: Text(
              entry.content,
              style: AppTypography.body(
                color: isCompleted ? colors.textMuted : colors.textBase,
              ).copyWith(
                decoration:
                    isCompleted ? TextDecoration.lineThrough : null,
              ),
            ),
          ),
        ],
      ),
    );
  }

  Widget _priorityBadge(PlanEntryPriority priority) {
    return switch (priority) {
      PlanEntryPriority.high =>
        const Badge('H', variant: BadgeVariant.error, size: BadgeSize.sm),
      PlanEntryPriority.medium =>
        const Badge('M', variant: BadgeVariant.warning, size: BadgeSize.sm),
      PlanEntryPriority.low =>
        const Badge('L', variant: BadgeVariant.neutral, size: BadgeSize.sm),
    };
  }

  IconData _iconForStatus(PlanEntryStatus status) {
    return switch (status) {
      PlanEntryStatus.completed => FluentIcons.check_mark,
      PlanEntryStatus.inProgress => FluentIcons.radio_bullet,
      PlanEntryStatus.pending => FluentIcons.circle_ring,
    };
  }
}
```

> Используй компоненты из `codelab_ui_components`:
>
> - `CollapsibleSectionHeader` — заменяет кастомный `InkWell`-заголовок (встроенная анимация chevron, toggle логика)
> - `AppProgressBar` — заменяет `LinearProgressIndicator` из Material
> - `Badge` с вариантами — заменяет кастомный `_PriorityBadge` (`BadgeVariant.error/warning/neutral`)
> - `AppIcon` — заменяет `Icon(icon, size: 16)` с токенами размера
> - `AppColors` / `AppTypography` / `AppSpacing` — токены дизайна

---

## Критерии готовности

- [ ] `PlanPanel` показывается только при наличии плана
- [ ] `CollapsibleSectionHeader` с прогрессом `N/M` и кнопкой свернуть/развернуть
- [ ] `AppProgressBar` отображает `completedCount / totalCount`
- [ ] `Badge` показывает приоритет (`H`/`M`/`L`) с цветом из дизайн-системы
- [ ] Завершённые пункты перечёркнуты через `TextDecoration.lineThrough`
