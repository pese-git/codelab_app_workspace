# Задача 6.2 — ChatBloc

## Цель

Реализовать BLoC для управления чатом: отправка промптов, streaming ответов, накопление tool calls и плана.

## Концепция

`ChatBloc` — аналог Python `ChatViewModel`. Принимает события от пользователя, запускает `SendPromptUseCase` и применяет streaming обновления к состоянию по мере их поступления.

---

## Шаг 1 — Events

Создай файл `lib/presentation/blocs/chat/chat_event.dart`:

```dart
import 'package:freezed_annotation/freezed_annotation.dart';

part 'chat_event.freezed.dart';

@freezed
sealed class ChatEvent with _$ChatEvent {
  const factory ChatEvent.sessionOpened({required String sessionId}) =
      ChatSessionOpened;

  const factory ChatEvent.promptSubmitted({required String text}) =
      ChatPromptSubmitted;

  const factory ChatEvent.promptCancelled() = ChatPromptCancelled;

  /// Внутреннее событие — получен session/update chunk
  const factory ChatEvent.updateReceived({
    required Map<String, dynamic> update,
  }) = ChatUpdateReceived;

  const factory ChatEvent.cleared() = ChatCleared;
}
```

## Шаг 2 — States

Создай файл `lib/presentation/blocs/chat/chat_state.dart`:

```dart
import 'package:freezed_annotation/freezed_annotation.dart';

import '../../../domain/entities/chat_message.dart';
import '../../../domain/entities/tool_call_record.dart';
import '../../../domain/entities/plan_state.dart';

part 'chat_state.freezed.dart';

@freezed
class ChatState with _$ChatState {
  const factory ChatState({
    required String sessionId,
    @Default([]) List<ChatMessage> messages,
    @Default({}) Map<String, ToolCallRecord> toolCalls,
    @Default(PlanState()) PlanState plan,
    @Default(false) bool isProcessing,
    String? errorMessage,
    String? streamingMessageId,
  }) = _ChatState;

  factory ChatState.initial(String sessionId) =>
      ChatState(sessionId: sessionId);

  const ChatState._();

  bool get hasMessages => messages.isNotEmpty;
  bool get hasToolCalls => toolCalls.isNotEmpty;
  bool get hasPlan => !plan.isEmpty;
}
```

## Шаг 3 — BLoC

Создай файл `lib/presentation/blocs/chat/chat_bloc.dart`:

```dart
import 'package:bloc/bloc.dart';
import 'package:structured_log/structured_log.dart';

import '../../../application/use_cases/send_prompt_use_case.dart';
import '../../../application/dto/session_dto.dart';
import '../../../domain/entities/chat_message.dart';
import '../../../domain/entities/tool_call_record.dart';
import '../../../domain/entities/plan_state.dart';
import '../../../domain/entities/session_update.dart';
import '../../../domain/entities/tool_call.dart';
import '../../../domain/entities/plan.dart';
import 'chat_event.dart';
import 'chat_state.dart';

class ChatBloc extends Bloc<ChatEvent, ChatState> {
  final _log = getLogger('ChatBloc');
  final SendPromptUseCase _sendPromptUseCase;

  ChatBloc({required SendPromptUseCase sendPromptUseCase})
      : _sendPromptUseCase = sendPromptUseCase,
        super(ChatState.initial('')) {
    on<ChatSessionOpened>(_onSessionOpened);
    on<ChatPromptSubmitted>(_onPromptSubmitted);
    on<ChatUpdateReceived>(_onUpdateReceived);
    on<ChatCleared>(_onCleared);
  }

  void _onSessionOpened(
    ChatSessionOpened event,
    Emitter<ChatState> emit,
  ) {
    emit(ChatState.initial(event.sessionId));
  }

  Future<void> _onPromptSubmitted(
    ChatPromptSubmitted event,
    Emitter<ChatState> emit,
  ) async {
    if (state.isProcessing) return;

    // Добавляем сообщение пользователя
    final userMessage = ChatMessage.userMessage(event.text);

    // Добавляем placeholder для ответа агента (streaming)
    final agentMessage = ChatMessage.agentMessageStreaming();

    emit(state.copyWith(
      messages: [...state.messages, userMessage, agentMessage],
      isProcessing: true,
      streamingMessageId: agentMessage.id,
      errorMessage: null,
    ));

    final result = await _sendPromptUseCase.execute(
      SendPromptRequestDto(
        sessionId: state.sessionId,
        promptText: event.text,
      ),
      callbacks: PromptCallbacks(
        onUpdate: (update) => add(ChatEvent.updateReceived(update: update)),
      ),
    );

    result.fold(
      (failure) {
        _log.error('Prompt failed: ${failure.message}');
        emit(state.copyWith(
          isProcessing: false,
          streamingMessageId: null,
          errorMessage: failure.message,
          // Завершаем streaming message с ошибкой
          messages: _finalizeStreamingMessage(state.messages, agentMessage.id),
        ));
      },
      (response) {
        emit(state.copyWith(
          isProcessing: false,
          streamingMessageId: null,
          // Финализируем streaming message
          messages: _finalizeStreamingMessage(state.messages, agentMessage.id),
        ));
      },
    );
  }

  void _onUpdateReceived(
    ChatUpdateReceived event,
    Emitter<ChatState> emit,
  ) {
    final update = event.update;
    final params = update['params'] as Map<String, dynamic>?;
    if (params == null) return;

    final updateData = params['update'] as Map<String, dynamic>?;
    if (updateData == null) return;

    final sessionUpdate = updateData['sessionUpdate'] as String?;

    switch (sessionUpdate) {
      case 'agent_message_chunk':
        _handleMessageChunk(updateData, emit);
      case 'agent_thought_chunk':
        _handleThoughtChunk(updateData, emit);
      case 'tool_call':
        _handleToolCallCreated(updateData, emit);
      case 'tool_call_update':
        _handleToolCallUpdate(updateData, emit);
      case 'plan':
        _handlePlanUpdate(updateData, emit);
    }
  }

  void _handleMessageChunk(
    Map<String, dynamic> data,
    Emitter<ChatState> emit,
  ) {
    final chunkUpdate = MessageChunkUpdate.fromJson(data);
    final text = chunkUpdate.text;
    if (text == null || text.isEmpty) return;

    final streamingId = state.streamingMessageId;
    if (streamingId == null) return;

    final updatedMessages = state.messages.map((msg) {
      if (msg.id == streamingId) {
        return msg.appendChunk(text);
      }
      return msg;
    }).toList();

    emit(state.copyWith(messages: updatedMessages));
  }

  void _handleThoughtChunk(
    Map<String, dynamic> data,
    Emitter<ChatState> emit,
  ) {
    // Thoughts можно показывать отдельно или игнорировать
    // Здесь просто логируем
    _log.debug('Thought chunk received');
  }

  void _handleToolCallCreated(
    Map<String, dynamic> data,
    Emitter<ChatState> emit,
  ) {
    final update = ToolCallCreatedUpdate.fromJson(data);
    final record = ToolCallRecord.fromCreatedUpdate(update);
    final newToolCalls = Map<String, ToolCallRecord>.from(state.toolCalls);
    newToolCalls[update.toolCallId] = record;
    emit(state.copyWith(toolCalls: newToolCalls));
  }

  void _handleToolCallUpdate(
    Map<String, dynamic> data,
    Emitter<ChatState> emit,
  ) {
    final update = ToolCallStateUpdate.fromJson(data);
    final existing = state.toolCalls[update.toolCallId];
    if (existing == null) return;

    final newToolCalls = Map<String, ToolCallRecord>.from(state.toolCalls);
    newToolCalls[update.toolCallId] = existing.applyUpdate(update);
    emit(state.copyWith(toolCalls: newToolCalls));
  }

  void _handlePlanUpdate(
    Map<String, dynamic> data,
    Emitter<ChatState> emit,
  ) {
    final planUpdate = PlanUpdate.fromJson(data);
    emit(state.copyWith(
      plan: state.plan.updateEntries(planUpdate.entries),
    ));
  }

  void _onCleared(ChatCleared event, Emitter<ChatState> emit) {
    emit(ChatState.initial(state.sessionId));
  }

  List<ChatMessage> _finalizeStreamingMessage(
    List<ChatMessage> messages,
    String messageId,
  ) {
    return messages.map((msg) {
      if (msg.id == messageId) return msg.complete();
      return msg;
    }).toList();
  }
}
```

## Критерии готовности

- [ ] `ChatEvent` sealed class реализован
- [ ] `ChatState` с иммутабельным списком messages, toolCalls, plan
- [ ] Streaming: `appendChunk()` вызывается при каждом `agent_message_chunk`
- [ ] Tool calls обновляются через `applyUpdate()`
- [ ] Plan обновляется через `updateEntries()`
- [ ] Ошибки отображаются в состоянии без краша
- [ ] `ChatBloc` регистрируется через `BlocProvider` в навигации с разрешением из `sessionScope`
