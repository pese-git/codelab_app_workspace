# Задача 5.3 — LoadSessionUseCase & ListSessionsUseCase

## Цель

Реализовать Use Cases для загрузки существующей сессии и получения списка сессий.

---

## Шаг 1 — ListSessionsUseCase

Создай файл `lib/application/use_cases/list_sessions_use_case.dart`:

```dart
import 'package:fpdart/fpdart.dart';
import 'package:structured_log/structured_log.dart';

import '../../core/error/failures.dart';
import '../../domain/entities/acp_message.dart';
import '../../domain/entities/session.dart';
import '../../domain/entities/session_list.dart';
import '../../domain/repositories/session_repository.dart';
import '../../domain/services/transport_service.dart';
import '../dto/session_dto.dart';

/// Use Case для получения списка сессий от ACP сервера
///
/// Аналог Python: application/use_cases.py::ListSessionsUseCase
/// Регистрируется в AppModule как transient.
class ListSessionsUseCase {
  final _log = getLogger('ListSessionsUseCase');
  final TransportService _transport;
  final SessionRepository _sessionRepo;

  ListSessionsUseCase({
    required TransportService transport,
    required SessionRepository sessionRepo,
  })  : _transport = transport,
        _sessionRepo = sessionRepo;

  Future<Either<Failure, ListSessionsResponseDto>> execute() async {
    _log.info('ListSessionsUseCase: requesting session list');

    try {
      if (!_transport.isInitialized()) {
        return left(SessionFailure(
          message: 'Transport not initialized.',
        ));
      }

      final responseData = await _transport.requestWithCallbacks(
        method: 'session/list',
        params: {},
      );

      final response = AcpMessage.fromJson(responseData);

      if (response.hasError) {
        return left(ProtocolFailure(
          message: 'session/list failed: ${response.error!.message}',
          errorCode: response.error!.code,
        ));
      }

      final result = SessionListResult.fromJson(
        response.result as Map<String, dynamic>,
      );

      // Синхронизируем локальный кэш сессий
      for (final item in result.sessions) {
        final cached = await _sessionRepo.load(item.sessionId);
        if (cached == null) {
          final session = Session.create(
            serverHost: 'unknown',
            serverPort: 0,
            clientCapabilities: {},
            serverCapabilities: _transport.getServerCapabilities(),
            sessionId: item.sessionId,
            cwd: item.cwd,
          );
          await _sessionRepo.save(session);
        }
      }

      final sessions = result.sessions
          .map((item) => SessionListItemDto(
                sessionId: item.sessionId,
                cwd: item.cwd,
                title: item.title,
                updatedAt: item.updatedAt,
              ))
          .toList();

      _log.info('Listed ${sessions.length} sessions');

      return right(ListSessionsResponseDto(sessions: sessions));
    } on Failure catch (f) {
      return left(f);
    } catch (e) {
      return left(UnexpectedFailure(message: 'ListSessions failed: $e'));
    }
  }
}
```

## Шаг 2 — LoadSessionUseCase

Создай файл `lib/application/use_cases/load_session_use_case.dart`:

```dart
import 'package:fpdart/fpdart.dart';
import 'package:structured_log/structured_log.dart';

import '../../core/error/failures.dart';
import '../../domain/entities/acp_message.dart';
import '../../domain/entities/session.dart';
import '../../domain/repositories/session_repository.dart';
import '../../domain/services/transport_service.dart';
import '../dto/session_dto.dart';

/// Use Case для загрузки (восстановления) существующей сессии
///
/// Отправляет session/load, получает replay updates для восстановления UI.
/// Аналог Python: application/use_cases.py::LoadSessionUseCase
/// Регистрируется в AppModule как transient.
class LoadSessionUseCase {
  final _log = getLogger('LoadSessionUseCase');
  final TransportService _transport;
  final SessionRepository _sessionRepo;

  LoadSessionUseCase({
    required TransportService transport,
    required SessionRepository sessionRepo,
  })  : _transport = transport,
        _sessionRepo = sessionRepo;

  Future<Either<Failure, LoadSessionResponseDto>> execute(
    LoadSessionRequestDto request,
  ) async {
    _log.info('LoadSessionUseCase: sessionId=${request.sessionId}');

    try {
      if (!_transport.isInitialized()) {
        return left(SessionFailure(
          message: 'Transport not initialized.',
        ));
      }

      // Загружаем или создаём shadow-копию в репозитории
      var session = await _sessionRepo.load(request.sessionId);
      if (session == null) {
        _log.debug('Session not in repo, creating shadow copy');
        session = Session.create(
          serverHost: request.serverHost,
          serverPort: request.serverPort,
          clientCapabilities: {},
          serverCapabilities: _transport.getServerCapabilities(),
          sessionId: request.sessionId,
        );
        await _sessionRepo.save(session);
      }

      // Собираем replay updates
      final replayUpdates = <Map<String, dynamic>>[];

      final responseData = await _transport.requestWithCallbacks(
        method: 'session/load',
        params: {
          'sessionId': request.sessionId,
          'cwd': request.cwd ?? '',
          'mcpServers': request.mcpServers ?? [],
        },
        onUpdate: (update) {
          // Фильтруем только обновления для этой сессии
          final params = update['params'] as Map<String, dynamic>?;
          if (params?['sessionId'] == request.sessionId) {
            replayUpdates.add(update);
          }
        },
      );

      final response = AcpMessage.fromJson(responseData);

      if (response.hasError) {
        return left(ProtocolFailure(
          message: 'session/load failed: ${response.error!.message}',
          errorCode: response.error!.code,
        ));
      }

      _log.info(
        'Session loaded: ${request.sessionId}, '
        'replay updates: ${replayUpdates.length}',
      );

      return right(LoadSessionResponseDto(
        sessionId: session.id,
        serverCapabilities: session.serverCapabilities,
        isAuthenticated: session.isAuthenticated,
        replayUpdates: replayUpdates,
      ));
    } on Failure catch (f) {
      return left(f);
    } catch (e) {
      return left(UnexpectedFailure(message: 'LoadSession failed: $e'));
    }
  }
}
```

- [ ] Shadow-копия сессии создаётся при cold-start
