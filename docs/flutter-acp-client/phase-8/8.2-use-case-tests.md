## Шаг 2 — Тест InitializeUseCase

Создай файл `test/application/use_cases/initialize_use_case_test.dart`:

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';

import 'package:acp_client/application/use_cases/initialize_use_case.dart';
import 'package:acp_client/application/dto/session_dto.dart';
import 'package:acp_client/domain/entities/acp_message.dart';

import '../../helpers/mocks.dart';

void main() {
  late MockTransportService transport;
  late InitializeUseCase useCase;

  setUp(() {
    transport = MockTransportService();
    useCase = InitializeUseCase(transport: transport);
  });

  group('InitializeUseCase', () {
    test('returns Right on successful initialize', () async {
      // Arrange
      when(() => transport.connect()).thenAnswer((_) async {});
      when(() => transport.send(any())).thenAnswer((_) async {});
      when(() => transport.receive(requestId: any(named: 'requestId')))
          .thenAnswer((_) async => AcpMessage.response('req_1', {
                'protocolVersion': 1,
                'agentCapabilities': {'loadSession': true},
                'authMethods': [],
              }).toJson());
      when(() => transport.setServerCapabilities(any())).thenReturn(null);

      // Act
      final result = await useCase.execute(
        const InitializeRequestDto(
          serverHost: 'localhost',
          serverPort: 8080,
        ),
      );

      // Assert
      expect(result.isRight(), true);
      result.fold(
        (_) => fail('Expected Right'),
        (response) {
          expect(response.protocolVersion, '1');
        },
      );
      verify(() => transport.connect()).called(1);
      verify(() => transport.setServerCapabilities(any())).called(1);
    });

    test('returns Left(ProtocolFailure) on server error', () async {
      when(() => transport.connect()).thenAnswer((_) async {});
      when(() => transport.send(any())).thenAnswer((_) async {});
      when(() => transport.receive(requestId: any(named: 'requestId')))
          .thenAnswer((_) async => AcpMessage.errorResponse(
                'req_1',
                code: -32600,
                message: 'Unsupported protocol',
              ).toJson());
      when(() => transport.disconnect()).thenAnswer((_) async {});

      final result = await useCase.execute(
        const InitializeRequestDto(
          serverHost: 'localhost',
          serverPort: 8080,
        ),
      );

      expect(result.isLeft(), true);
      result.fold(
        (failure) => expect(failure.message, contains('Unsupported protocol')),
        (_) => fail('Expected Left'),
      );
    });
  });
}
```

## Шаг 3 — Тест SendPromptUseCase

Создай файл `test/application/use_cases/send_prompt_use_case_test.dart`:

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';

import 'package:acp_client/application/use_cases/send_prompt_use_case.dart';
import 'package:acp_client/application/dto/session_dto.dart';
import 'package:acp_client/domain/entities/acp_message.dart';
import 'package:acp_client/domain/entities/session.dart';

import '../../helpers/mocks.dart';

void main() {
  late MockTransportService transport;
  late MockSessionRepository sessionRepo;
  late MockFileSystemService fsService;
  late MockTerminalService terminalService;
  late SendPromptUseCase useCase;

  setUp(() {
    transport = MockTransportService();
    sessionRepo = MockSessionRepository();
    fsService = MockFileSystemService();
    terminalService = MockTerminalService();
    useCase = SendPromptUseCase(
      transport: transport,
      sessionRepo: sessionRepo,
      fsService: fsService,
      terminalService: terminalService,
    );
  });

  group('SendPromptUseCase', () {
    test('returns Left(SessionFailure) if session not found', () async {
      when(() => sessionRepo.load('unknown')).thenAnswer((_) async => null);

      final result = await useCase.execute(
        const SendPromptRequestDto(
          sessionId: 'unknown',
          promptText: 'Hello',
        ),
      );

      expect(result.isLeft(), true);
    });

    test('returns Right with stopReason on success', () async {
      final session = Session.create(
        serverHost: 'localhost',
        serverPort: 8080,
        clientCapabilities: {},
        serverCapabilities: {},
        sessionId: 'sess_1',
      );
      when(() => sessionRepo.load('sess_1')).thenAnswer((_) async => session);
      when(() => transport.requestWithCallbacks(
            method: any(named: 'method'),
            params: any(named: 'params'),
            onUpdate: any(named: 'onUpdate'),
            onFsRead: any(named: 'onFsRead'),
            onFsWrite: any(named: 'onFsWrite'),
            onTerminalCreate: any(named: 'onTerminalCreate'),
            onTerminalOutput: any(named: 'onTerminalOutput'),
            onTerminalWait: any(named: 'onTerminalWait'),
            onTerminalRelease: any(named: 'onTerminalRelease'),
            onTerminalKill: any(named: 'onTerminalKill'),
          )).thenAnswer((_) async =>
          AcpMessage.response('req_1', {'stopReason': 'end_turn'}).toJson());

      final result = await useCase.execute(
        const SendPromptRequestDto(
          sessionId: 'sess_1',
          promptText: 'Hello',
        ),
      );

      expect(result.isRight(), true);
      result.fold(
        (_) => fail('Expected Right'),
        (response) => expect(
          response.promptResult['stopReason'],
          'end_turn',
        ),
      );
    });
  });
}
```
