## Шаг 1 — Тесты AcpMessage

Создай файл `test/domain/entities/acp_message_test.dart`:

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:acp_client/domain/entities/acp_message.dart';

void main() {
  group('AcpMessage', () {
    test('creates request with auto-generated id', () {
      final msg = AcpMessage.request('initialize', params: {'version': 1});
      expect(msg.method, 'initialize');
      expect(msg.id, isNotNull);
      expect(msg.params, {'version': 1});
      expect(msg.isRequest, true);
    });

    test('creates notification without id', () {
      final msg = AcpMessage.notification('session/cancel', params: {'sessionId': 's1'});
      expect(msg.id, isNull);
      expect(msg.isNotification, true);
    });

    test('creates response with result', () {
      final msg = AcpMessage.response('req_1', {'sessionId': 'sess_1'});
      expect(msg.id, 'req_1');
      expect(msg.result, {'sessionId': 'sess_1'});
      expect(msg.isResponse, true);
      expect(msg.hasError, false);
    });

    test('creates error response', () {
      final msg = AcpMessage.errorResponse('req_1', code: -32600, message: 'Invalid');
      expect(msg.hasError, true);
      expect(msg.error!.code, -32600);
    });

    test('fromJson parses request correctly', () {
      final json = {
        'jsonrpc': '2.0',
        'id': 'abc',
        'method': 'session/list',
        'params': {},
      };
      final msg = AcpMessage.fromJson(json);
      expect(msg.method, 'session/list');
      expect(msg.id, 'abc');
    });

    test('fromJson parses response correctly', () {
      final json = {
        'jsonrpc': '2.0',
        'id': 'abc',
        'result': {'sessions': []},
      };
      final msg = AcpMessage.fromJson(json);
      expect(msg.isResponse, true);
      expect(msg.result, {'sessions': []});
    });

    test('toJson produces correct wire format for request', () {
      final msg = AcpMessage.request('initialize',
          params: {'v': 1}, requestId: 'req_1');
      final json = msg.toJson();
      expect(json['jsonrpc'], '2.0');
      expect(json['id'], 'req_1');
      expect(json['method'], 'initialize');
      expect(json['params'], {'v': 1});
      expect(json.containsKey('result'), false);
    });

    test('toJson for notification has no id', () {
      final msg = AcpMessage.notification('ping');
      final json = msg.toJson();
      expect(json.containsKey('id'), false);
      expect(json['method'], 'ping');
    });
  });
}
```

## Шаг 2 — Тесты Session Entity

Создай файл `test/domain/entities/session_test.dart`:

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:acp_client/domain/entities/session.dart';

void main() {
  group('Session', () {
    test('create() generates unique IDs', () {
      final s1 = Session.create(
        serverHost: 'localhost',
        serverPort: 8080,
        clientCapabilities: {},
        serverCapabilities: {},
      );
      final s2 = Session.create(
        serverHost: 'localhost',
        serverPort: 8080,
        clientCapabilities: {},
        serverCapabilities: {},
      );
      expect(s1.id, isNot(s2.id));
    });

    test('create() uses provided sessionId', () {
      final session = Session.create(
        serverHost: 'localhost',
        serverPort: 8080,
        clientCapabilities: {},
        serverCapabilities: {},
        sessionId: 'my-session',
      );
      expect(session.id, 'my-session');
    });

    test('isAuthenticated defaults to false', () {
      final session = Session.create(
        serverHost: 'localhost',
        serverPort: 8080,
        clientCapabilities: {},
        serverCapabilities: {},
      );
      expect(session.isAuthenticated, false);
    });
  });
}
```

## Шаг 3 — Тесты ChatMessage

Создай файл `test/domain/entities/chat_message_test.dart`:

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:acp_client/domain/entities/chat_message.dart';

void main() {
  group('ChatMessage', () {
    test('appendChunk accumulates text', () {
      final msg = ChatMessage.agentMessageStreaming();
      final updated = msg.appendChunk('Hello ').appendChunk('World');
      expect(updated.text, 'Hello World');
    });

    test('complete() changes status', () {
      final msg = ChatMessage.agentMessageStreaming();
      expect(msg.status, ChatMessageStatus.streaming);
      final completed = msg.complete();
      expect(completed.status, ChatMessageStatus.complete);
    });

    test('userMessage has correct role', () {
      final msg = ChatMessage.userMessage('test');
      expect(msg.role, ChatMessageRole.user);
      expect(msg.status, ChatMessageStatus.complete);
    });
  });
}
```
