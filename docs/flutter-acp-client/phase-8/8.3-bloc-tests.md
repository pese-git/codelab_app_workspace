## Шаг 1 — SessionBloc тесты

Создай файл `test/presentation/blocs/session_bloc_test.dart`:

```dart
import 'package:bloc_test/bloc_test.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:fpdart/fpdart.dart';
import 'package:mocktail/mocktail.dart';

import 'package:acp_client/presentation/blocs/session/session_bloc.dart';
import 'package:acp_client/presentation/blocs/session/session_event.dart';
import 'package:acp_client/presentation/blocs/session/session_state.dart';
import 'package:acp_client/application/dto/session_dto.dart';
import 'package:acp_client/core/error/failures.dart';

class MockInitializeUseCase extends Mock {
  Future<Either<Failure, InitializeResponseDto>> execute(
    InitializeRequestDto request,
  );
}

class MockListSessionsUseCase extends Mock {
  Future<Either<Failure, ListSessionsResponseDto>> execute();
}

class MockCreateSessionUseCase extends Mock {}
class MockLoadSessionUseCase extends Mock {}
class MockTransportService extends Mock {
  Future<void> disconnect();
}

void main() {
  late MockInitializeUseCase initUseCase;
  late MockListSessionsUseCase listUseCase;
  late MockCreateSessionUseCase createUseCase;
  late MockLoadSessionUseCase loadUseCase;
  late MockTransportService transport;

  setUp(() {
    initUseCase = MockInitializeUseCase();
    listUseCase = MockListSessionsUseCase();
    createUseCase = MockCreateSessionUseCase();
    loadUseCase = MockLoadSessionUseCase();
    transport = MockTransportService();
  });

  SessionBloc buildBloc() => SessionBloc(
    initializeUseCase: initUseCase as dynamic,
    listSessionsUseCase: listUseCase as dynamic,
    createSessionUseCase: createUseCase as dynamic,
    loadSessionUseCase: loadUseCase as dynamic,
    transport: transport as dynamic,
  );

  group('SessionBloc', () {
    blocTest<SessionBloc, SessionState>(
      'emits [loading, initialized, loading, sessionListLoaded] on successful initialize',
      build: buildBloc,
      setUp: () {
        when(() => initUseCase.execute(any())).thenAnswer(
          (_) async => right(const InitializeResponseDto(
            serverCapabilities: {},
            availableAuthMethods: [],
            protocolVersion: '1',
          )),
        );
        when(() => listUseCase.execute()).thenAnswer(
          (_) async => right(const ListSessionsResponseDto(sessions: [])),
        );
      },
      act: (bloc) => bloc.add(const SessionEvent.initialize(
        serverHost: 'localhost',
        serverPort: 8080,
      )),
      expect: () => [
        isA<SessionLoading>(),
        isA<SessionInitialized>(),
        isA<SessionLoading>(),
        isA<SessionListLoaded>(),
      ],
    );

    blocTest<SessionBloc, SessionState>(
      'emits [loading, error] on initialize failure',
      build: buildBloc,
      setUp: () {
        when(() => initUseCase.execute(any())).thenAnswer(
          (_) async => left(
            const TransportFailure(message: 'Connection refused'),
          ),
        );
      },
      act: (bloc) => bloc.add(const SessionEvent.initialize(
        serverHost: 'localhost',
        serverPort: 8080,
      )),
      expect: () => [
        isA<SessionLoading>(),
        isA<SessionError>(),
      ],
    );
  });
}
```

## Шаг 2 — ChatBloc тесты

Создай файл `test/presentation/blocs/chat_bloc_test.dart`:

```dart
import 'package:bloc_test/bloc_test.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:fpdart/fpdart.dart';
import 'package:mocktail/mocktail.dart';

import 'package:acp_client/presentation/blocs/chat/chat_bloc.dart';
import 'package:acp_client/presentation/blocs/chat/chat_event.dart';
import 'package:acp_client/presentation/blocs/chat/chat_state.dart';
import 'package:acp_client/application/dto/session_dto.dart';
import 'package:acp_client/core/error/failures.dart';

class MockSendPromptUseCase extends Mock {
  Future<Either<Failure, SendPromptResponseDto>> execute(
    SendPromptRequestDto request, {
    dynamic callbacks,
  });
}

void main() {
  late MockSendPromptUseCase sendPromptUseCase;

  setUp(() {
    sendPromptUseCase = MockSendPromptUseCase();
  });

  ChatBloc buildBloc() => ChatBloc(
    sendPromptUseCase: sendPromptUseCase as dynamic,
  );

  group('ChatBloc', () {
    blocTest<ChatBloc, ChatState>(
      'sessionOpened initializes state with sessionId',
      build: buildBloc,
      act: (bloc) => bloc.add(
        const ChatEvent.sessionOpened(sessionId: 'sess_1'),
      ),
      expect: () => [
        isA<ChatState>().having((s) => s.sessionId, 'sessionId', 'sess_1'),
      ],
    );

    blocTest<ChatBloc, ChatState>(
      'promptSubmitted adds user message and sets isProcessing',
      build: () {
        final bloc = buildBloc();
        bloc.add(const ChatEvent.sessionOpened(sessionId: 'sess_1'));
        return bloc;
      },
      setUp: () {
        when(() => sendPromptUseCase.execute(
          any(),
          callbacks: any(named: 'callbacks'),
        )).thenAnswer((_) async => right(const SendPromptResponseDto(
          sessionId: 'sess_1',
          promptResult: {'stopReason': 'end_turn'},
        )));
      },
      act: (bloc) => bloc.add(
        const ChatEvent.promptSubmitted(text: 'Hello'),
      ),
      expect: () => [
        // После sessionOpened
        isA<ChatState>().having((s) => s.sessionId, 'sessionId', 'sess_1'),
        // После promptSubmitted: сообщения добавились, isProcessing=true
        isA<ChatState>()
            .having((s) => s.messages.length, 'messages', greaterThan(0))
            .having((s) => s.isProcessing, 'isProcessing', true),
        // После завершения: isProcessing=false
        isA<ChatState>().having((s) => s.isProcessing, 'isProcessing', false),
      ],
    );
  });
}
```
