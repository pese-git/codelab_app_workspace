# Задача 8.4 — Widget Tests

## Цель

Widget-тесты для кастомных UI-компонентов: `ToolCallCard`, `PlanPanel`, `PermissionDialog`.

> **Важно:** `MessageBubble` и `MessageTimeline` — компоненты из `codelab_ui_components`, их тестировать в этом проекте не нужно. Тестируем только виджеты, созданные в `lib/presentation/widgets/`.

---

## Шаг 1 — ToolCallCard тест

Создай файл `test/presentation/widgets/tool_call_card_test.dart`:

```dart
import 'package:fluent_ui/fluent_ui.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:codelab_ui_components/codelab_ui_components.dart';

import 'package:acp_client/presentation/widgets/tool_call_card.dart';
import 'package:acp_client/domain/entities/tool_call_record.dart';
import 'package:acp_client/domain/entities/tool_call.dart';

void main() {
  Widget buildTestWidget(Widget child) {
    return FluentApp(
      home: ScaffoldPage(content: child),
    );
  }

  group('ToolCallCard', () {
    testWidgets('shows title', (tester) async {
      final toolCall = ToolCallRecord(
        toolCallId: 'tc_1',
        title: 'Read config.yaml',
        kind: ToolKind.read,
        status: ToolCallStatus.completed,
      );
      await tester.pumpWidget(
        buildTestWidget(ToolCallCard(toolCall: toolCall)),
      );
      expect(find.text('Read config.yaml'), findsOneWidget);
    });

    testWidgets('shows AppProgressRing for in-progress status', (tester) async {
      final toolCall = ToolCallRecord(
        toolCallId: 'tc_2',
        title: 'Write file',
        kind: ToolKind.edit,
        status: ToolCallStatus.inProgress,
      );
      await tester.pumpWidget(
        buildTestWidget(ToolCallCard(toolCall: toolCall)),
      );
      expect(find.byType(AppProgressRing), findsOneWidget);
    });

    testWidgets('shows success Badge for completed status', (tester) async {
      final toolCall = ToolCallRecord(
        toolCallId: 'tc_3',
        title: 'Done',
        kind: ToolKind.read,
        status: ToolCallStatus.completed,
      );
      await tester.pumpWidget(
        buildTestWidget(ToolCallCard(toolCall: toolCall)),
      );
      expect(find.text('Done'), findsOneWidget);
    });
  });
}
```

## Шаг 2 — PlanPanel тест

Создай файл `test/presentation/widgets/plan_panel_test.dart`:

```dart
import 'package:fluent_ui/fluent_ui.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:codelab_ui_components/codelab_ui_components.dart';

import 'package:acp_client/presentation/widgets/plan_panel.dart';
import 'package:acp_client/domain/entities/plan_state.dart';
import 'package:acp_client/domain/entities/plan.dart';

void main() {
  Widget buildTestWidget(Widget child) {
    return FluentApp(
      home: ScaffoldPage(content: child),
    );
  }

  group('PlanPanel', () {
    testWidgets('shows plan entries', (tester) async {
      final plan = PlanState.empty().updateEntries([
        const PlanEntry(
          content: 'Analyze codebase',
          priority: PlanEntryPriority.high,
          status: PlanEntryStatus.completed,
        ),
        const PlanEntry(
          content: 'Write tests',
          priority: PlanEntryPriority.medium,
          status: PlanEntryStatus.inProgress,
        ),
      ]);

      await tester.pumpWidget(
        buildTestWidget(PlanPanel(plan: plan)),
      );

      expect(find.text('Analyze codebase'), findsOneWidget);
      expect(find.text('Write tests'), findsOneWidget);
    });

    testWidgets('shows CollapsibleSectionHeader with progress subtitle', (tester) async {
      final plan = PlanState.empty().updateEntries([
        const PlanEntry(
          content: 'Step 1',
          priority: PlanEntryPriority.high,
          status: PlanEntryStatus.completed,
        ),
        const PlanEntry(
          content: 'Step 2',
          priority: PlanEntryPriority.medium,
          status: PlanEntryStatus.pending,
        ),
      ]);

      await tester.pumpWidget(
        buildTestWidget(PlanPanel(plan: plan)),
      );

      expect(find.byType(CollapsibleSectionHeader), findsOneWidget);
      expect(find.text('1/2'), findsOneWidget);
    });

    testWidgets('uses AppProgressBar', (tester) async {
      final plan = PlanState.empty().updateEntries([
        const PlanEntry(
          content: 'Step',
          priority: PlanEntryPriority.low,
          status: PlanEntryStatus.inProgress,
        ),
      ]);

      await tester.pumpWidget(
        buildTestWidget(PlanPanel(plan: plan)),
      );

      expect(find.byType(AppProgressBar), findsOneWidget);
    });
  });
}
```
