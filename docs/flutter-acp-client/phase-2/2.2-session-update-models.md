# Задача 2.2 — Session Update типы и Content блоки

## Цель

Реализовать все типы `session/update` уведомлений и Content блоки ACP протокола.

## Концепция

`session/update` — это notification от агента к клиенту, который несёт разные типы данных в зависимости от поля `sessionUpdate`:

| `sessionUpdate` | Назначение |
|---|---|
| `agent_message_chunk` | Чанк текста ответа агента |
| `user_message_chunk` | Чанк текста сообщения пользователя |
| `agent_thought_chunk` | Размышления агента (reasoning) |
| `tool_call` | Новый вызов инструмента |
| `tool_call_update` | Обновление статуса вызова |
| `plan` | Обновление плана выполнения |
| `session_info_update` | Метаданные сессии (title и т.д.) |
| `current_mode_update` | Смена режима агента |
| `available_commands_update` | Список slash-команд |
| `config_option_update` | Обновление настроек |

@freezed
class ResourceLinkContentBlock with _$ResourceLinkContentBlock {
  const factory ResourceLinkContentBlock({
    @Default('resource_link') String type,
    required String uri,
    required String name,
  }) = _ResourceLinkContentBlock;

  factory ResourceLinkContentBlock.fromJson(Map<String, dynamic> json) =>
      _$ResourceLinkContentBlockFromJson(json);

/// Встроенный ресурс
@freezed
class EmbeddedResourceContentBlock with _$EmbeddedResourceContentBlock {
  const factory EmbeddedResourceContentBlock({
    @Default('resource') String type,
    required Map<String, dynamic> resource,
  }) = _EmbeddedResourceContentBlock;

  factory EmbeddedResourceContentBlock.fromJson(Map<String, dynamic> json) =>
      _$EmbeddedResourceContentBlockFromJson(json);
}

/// Union тип для Content блоков
sealed class ContentBlock {
  const ContentBlock();

  factory ContentBlock.fromJson(Map<String, dynamic> json) {
    return switch (json['type'] as String?) {
      'text' => TextContentBlock.fromJson(json),
      'image' => ImageContentBlock.fromJson(json),
      'audio' => AudioContentBlock.fromJson(json),
      'resource_link' => ResourceLinkContentBlock.fromJson(json),
      'resource' => EmbeddedResourceContentBlock.fromJson(json),
      _ => TextContentBlock(text: json.toString()),
    };
  }
}

## Шаг 2 — Session Update типы

Создай файл `lib/domain/entities/session_update.dart`:

```dart
import 'package:freezed_annotation/freezed_annotation.dart';

import 'content_block.dart';

part 'session_update.freezed.dart';
part 'session_update.g.dart';

/// Message chunk (agent или user)
@freezed
class MessageChunkUpdate with _$MessageChunkUpdate {
  const factory MessageChunkUpdate({
    required String sessionUpdate, // 'agent_message_chunk' | 'user_message_chunk'
    required Map<String, dynamic> content,
  }) = _MessageChunkUpdate;

  factory MessageChunkUpdate.fromJson(Map<String, dynamic> json) =>
      _$MessageChunkUpdateFromJson(json);

  const MessageChunkUpdate._();

  bool get isAgent => sessionUpdate == 'agent_message_chunk';

  /// Возвращает текст если content — текстовый блок
  String? get text {
    if (content['type'] == 'text') {
      return content['text'] as String?;
    }
    return null;
  }
}

/// Reasoning chunk агента
@freezed
class ThoughtChunkUpdate with _$ThoughtChunkUpdate {
  const factory ThoughtChunkUpdate({
    @Default('agent_thought_chunk') String sessionUpdate,
    required Map<String, dynamic> content,
  }) = _ThoughtChunkUpdate;

  factory ThoughtChunkUpdate.fromJson(Map<String, dynamic> json) =>
      _$ThoughtChunkUpdateFromJson(json);

  const ThoughtChunkUpdate._();

  String? get text {
    if (content['type'] == 'text') {
      return content['text'] as String?;
    }
    return null;
  }
}

/// Обновление метаданных сессии
@freezed
class SessionInfoUpdate with _$SessionInfoUpdate {
  const factory SessionInfoUpdate({
    @Default('session_info_update') String sessionUpdate,
    String? title,
    String? updatedAt,
  }) = _SessionInfoUpdate;

  factory SessionInfoUpdate.fromJson(Map<String, dynamic> json) =>
      _$SessionInfoUpdateFromJson(json);
}

/// Смена режима агента
@freezed
class CurrentModeUpdate with _$CurrentModeUpdate {
  const factory CurrentModeUpdate({
    @Default('current_mode_update') String sessionUpdate,
    required String currentModeId,
  }) = _CurrentModeUpdate;

  factory CurrentModeUpdate.fromJson(Map<String, dynamic> json) =>
      _$CurrentModeUpdateFromJson(json);
}

/// Доступные slash-команды
@freezed
class AvailableCommand with _$AvailableCommand {
  const factory AvailableCommand({
    required String name,
    required String description,
    Map<String, dynamic>? input,
  }) = _AvailableCommand;

  factory AvailableCommand.fromJson(Map<String, dynamic> json) =>
      _$AvailableCommandFromJson(json);
}

@freezed
class AvailableCommandsUpdate with _$AvailableCommandsUpdate {
  const factory AvailableCommandsUpdate({
    @Default('available_commands_update') String sessionUpdate,
    required List<AvailableCommand> availableCommands,
  }) = _AvailableCommandsUpdate;

  factory AvailableCommandsUpdate.fromJson(Map<String, dynamic> json) =>
      _$AvailableCommandsUpdateFromJson(json);
}

/// Базовый payload любого session/update
class SessionUpdatePayload {
  final String sessionUpdate;
  final Map<String, dynamic> raw;

  const SessionUpdatePayload({
    required this.sessionUpdate,
    required this.raw,
  });

  factory SessionUpdatePayload.fromJson(Map<String, dynamic> json) {
    return SessionUpdatePayload(
      sessionUpdate: json['sessionUpdate'] as String,
      raw: json,
    );
  }
}

/// Параметры notification session/update
class SessionUpdateParams {
  final String sessionId;
  final SessionUpdatePayload update;

  const SessionUpdateParams({
    required this.sessionId,
    required this.update,
  });

  factory SessionUpdateParams.fromJson(Map<String, dynamic> json) {
    return SessionUpdateParams(
      sessionId: json['sessionId'] as String,
      update: SessionUpdatePayload.fromJson(
        json['update'] as Map<String, dynamic>,
      ),
    );
  }
}

/// Парсер для разбора session/update payload в конкретные типы
class SessionUpdateParser {
  /// Парсит raw payload в один из типизированных update
  Object? parse(SessionUpdatePayload payload) {
    return switch (payload.sessionUpdate) {
      'agent_message_chunk' ||
      'user_message_chunk' =>
        MessageChunkUpdate.fromJson(payload.raw),
      'agent_thought_chunk' => ThoughtChunkUpdate.fromJson(payload.raw),
      'session_info_update' => SessionInfoUpdate.fromJson(payload.raw),
      'current_mode_update' => CurrentModeUpdate.fromJson(payload.raw),
      'available_commands_update' =>
        AvailableCommandsUpdate.fromJson(payload.raw),
      'tool_call' => null, // обрабатывается в task 2.3
      'tool_call_update' => null, // обрабатывается в task 2.3
      'plan' => null, // обрабатывается в task 2.3
      _ => null,
    };
  }
}
```

## Шаг 3 — Prompt Result

Добавь модель в файл `lib/domain/entities/prompt_result.dart`:

```dart
import 'package:freezed_annotation/freezed_annotation.dart';

part 'prompt_result.freezed.dart';
part 'prompt_result.g.dart';

typedef StopReason = String;
// Возможные значения:
// 'end_turn' | 'max_tokens' | 'max_turn_requests' | 'refusal' | 'cancelled'

@freezed
class PromptResult with _$PromptResult {
  const factory PromptResult({
    required String stopReason,
  }) = _PromptResult;

  factory PromptResult.fromJson(Map<String, dynamic> json) =>
      _$PromptResultFromJson(json);

  const PromptResult._();

  bool get isEndTurn => stopReason == 'end_turn';
  bool get isCancelled => stopReason == 'cancelled';
}
```
