# Задача 2.1 — ACP Message Models (JSON-RPC 2.0)

## Цель

Реализовать типизированные модели JSON-RPC 2.0 сообщений ACP протокола. Это Flutter-аналог `messages.py` из референса.

## Концепция

ACP протокол использует JSON-RPC 2.0. Каждое сообщение — это одна из трёх форм:
- **Request** — запрос с `method`, `params`, `id`
- **Notification** — уведомление с `method`, `params`, без `id`
- **Response** — ответ с `id` и `result` или `error`

В Dart используем `freezed` для иммутабельных моделей и `json_serializable` для сериализации.

## Шаг 1 — Базовые JSON-RPC типы

Создай файл `lib/domain/entities/acp_message.dart`:

```dart
import 'package:freezed_annotation/freezed_annotation.dart';
import 'package:uuid/uuid.dart';

part 'acp_message.freezed.dart';
part 'acp_message.g.dart';

const _uuid = Uuid();

// Тип для ID (строка или число)
typedef JsonRpcId = Object; // String или int

/// Ошибка JSON-RPC, полученная от сервера
@freezed
class JsonRpcError with _$JsonRpcError {
  const factory JsonRpcError({
    required int code,
    required String message,
    Object? data,
  }) = _JsonRpcError;

  factory JsonRpcError.fromJson(Map<String, dynamic> json) =>
      _$JsonRpcErrorFromJson(json);
}

/// Унифицированная модель JSON-RPC сообщения
/// Поддерживает request, notification и response формы
class AcpMessage {
  static const String jsonrpcVersion = '2.0';

  final String jsonrpc;
  final JsonRpcId? id;
  final String? method;
  final Map<String, dynamic>? params;
  final Object? result;
  final JsonRpcError? error;

  const AcpMessage._({
    this.jsonrpc = jsonrpcVersion,
    this.id,
    this.method,
    this.params,
    this.result,
    this.error,
  });

  /// Создаёт request сообщение с автогенерацией ID
  factory AcpMessage.request(
    String method, {
    Map<String, dynamic>? params,
    JsonRpcId? requestId,
  }) {
    return AcpMessage._(
      id: requestId ?? _uuid.v4().substring(0, 8),
      method: method,
      params: params ?? {},
    );
  }

  /// Создаёт notification сообщение (без ID)
  factory AcpMessage.notification(
    String method, {
    Map<String, dynamic>? params,
  }) {
    return AcpMessage._(
      id: null,
      method: method,
      params: params ?? {},
    );
  }

  /// Создаёт успешный response
  factory AcpMessage.response(JsonRpcId? requestId, Object? result) {
    return AcpMessage._(
      id: requestId,
      result: result,
    );
  }

  /// Создаёт error response
  factory AcpMessage.errorResponse(
    JsonRpcId? requestId, {
    required int code,
    required String message,
    Object? data,
  }) {
    return AcpMessage._(
      id: requestId,
      error: JsonRpcError(code: code, message: message, data: data),
    );
  }

  /// Десериализует JSON в AcpMessage
  factory AcpMessage.fromJson(Map<String, dynamic> json) {
    final error = json['error'] != null
        ? JsonRpcError.fromJson(json['error'] as Map<String, dynamic>)
        : null;

    return AcpMessage._(
      jsonrpc: json['jsonrpc'] as String? ?? jsonrpcVersion,
      id: json['id'] as JsonRpcId?,
      method: json['method'] as String?,
      params: json['params'] as Map<String, dynamic>?,
      result: json['result'],
      error: error,
    );
  }

  /// Сериализует в JSON (wire формат)
  Map<String, dynamic> toJson() {
    final payload = <String, dynamic>{'jsonrpc': jsonrpc};

    if (method != null) {
      // Request или Notification
      if (id != null) payload['id'] = id;
      payload['method'] = method;
      if (params != null) payload['params'] = params;
      return payload;
    }

    // Response
    payload['id'] = id;
    if (error != null) {
      payload['error'] = {
        'code': error!.code,
        'message': error!.message,
        if (error!.data != null) 'data': error!.data,
      };
    } else {
      payload['result'] = result;
    }
    return payload;
  }

  /// Проверяет, является ли сообщение request-ом
  bool get isRequest => method != null && id != null;

  /// Проверяет, является ли сообщение notification-ом
  bool get isNotification => method != null && id == null;

  /// Проверяет, является ли сообщение response-ом
  bool get isResponse => method == null;

  /// Проверяет наличие ошибки в response
  bool get hasError => error != null;

  @override
  String toString() => 'AcpMessage(id: $id, method: $method)';
}
```

## Шаг 6 — Парсинг утилиты

Создай файл `lib/infrastructure/transport/message_parser.dart`:

```dart
import '../../domain/entities/acp_message.dart';
import '../../domain/entities/initialize_result.dart';
import '../../domain/entities/session_list.dart';
import '../../domain/entities/session_setup.dart';
import '../../core/error/failures.dart';

/// Парсит AcpMessage в типизированные результаты
class AcpMessageParser {
  InitializeResult parseInitializeResult(AcpMessage message) {
    _assertNoError(message, 'initialize');
    _assertResultIsMap(message, 'initialize');
    return InitializeResult.fromJson(
      message.result as Map<String, dynamic>,
    );
  }

  SessionListResult parseSessionListResult(AcpMessage message) {
    _assertNoError(message, 'session/list');
    _assertResultIsMap(message, 'session/list');
    return SessionListResult.fromJson(
      message.result as Map<String, dynamic>,
    );
  }

  SessionSetupResult parseSessionSetupResult(
    AcpMessage message, {
    required String methodName,
  }) {
    _assertNoError(message, methodName);
    _assertResultIsMap(message, methodName);
    return SessionSetupResult.fromJson(
      message.result as Map<String, dynamic>,
    );
  }

  void _assertNoError(AcpMessage message, String method) {
    if (message.hasError) {
      throw ProtocolFailure(
        message: '$method failed: ${message.error!.message}',
        errorCode: message.error!.code,
      );
    }
  }

  void _assertResultIsMap(AcpMessage message, String method) {
    if (message.result is! Map<String, dynamic>) {
      throw ProtocolFailure(
        message: '$method response must contain object result',
      );
    }
  }
}
```
